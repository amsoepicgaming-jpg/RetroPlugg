cmake_minimum_required(VERSION 3.22)
project(RetroPlugg VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(RetroPlugg COMPANY_NAME "JW Audio" BUNDLE_ID "com.jwaudio.retroplugg" PLUGIN_MANUFACTURER_CODE JwAu PLUGIN_CODE RPlg FORMATS VST3 Standalone PRODUCT_NAME "RetroPlugg" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD TRUE)
target_sources(RetroPlugg PRIVATE Source/PluginProcessor.cpp Source/PluginProcessor.h Source/PluginEditor.cpp Source/PluginEditor.h)
target_link_libraries(RetroPlugg PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_audio_processors juce::juce_gui_extra)
target_compile_definitions(RetroPlugg PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
